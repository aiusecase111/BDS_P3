web-deployment.yaml
----------------------------
apiVersion: apps/v1
kind: Deployment
metadata:
  name: web-deployment
spec:
  replicas: 3
  selector:
    matchLabels:
      app: web
  template:
    metadata:
      labels:
        app: web
    spec:
      containers:
      - name: web
        image: nginx:alpine
        command: ["/bin/sh", "-c"]
        args:
        - |
          echo "Hello from Kubernetes Pod: $(hostname)" > /usr/share/nginx/html/index.html
          nginx -g 'daemon off;'
        ports:
        - containerPort: 80

-----------------------------------------------
web-service.yaml
-------------------------------------------------
apiVersion: v1
kind: Service
metadata:
  name: web-service
spec:
  type: ClusterIP
  selector:
    app: web
  ports:
  - port: 80
    targetPort: 80
